# Ключові слова для пошуку потенційних клієнтів.
# Файл можна редагувати під час роботи — програма перечитає його сама.
#
# Правила:
#   * одне слово або фраза на рядок; регістр не важливий;
#   * шукаються цілі слова: "fire" НЕ спрацює на "fireplace";
#   * зірочка в кінці — будь-яке закінчення: "plumb*" = plumber, plumbing;
#   * рядки, що починаються з #, — коментарі.
#
# Допис стає лідом, якщо містить хоча б одне слово з [include]
# і жодного слова з [exclude].
# ПОКИ [include] ПОРОЖНІЙ — КОЖЕН НОВИЙ ДОПИС ВВАЖАЄТЬСЯ ЛІДОМ (зручно для тестів).

[include]
# Приклади (приберіть # на початку рядка, щоб увімкнути):
# fire stopping
# fire stopper*
# looking for
# anyone recommend*

[exclude]
# Приклади:
# we are hiring
# drop us a message
